package handlers

import (
	"github.com/labstack/echo/v4"
)

type Router struct {
	articleHandler *ArticleHandler
}

func NewRouter(h *ArticleHandler) *Router {
	return &Router{articleHandler: h}
}

func (r *Router) Register(e *echo.Echo) {
	e.POST("/articles", r.articleHandler.CreateArticle)
	e.GET("/articles/:id", r.articleHandler.GetArticle)             // Slice 1
	e.PUT("/articles/:id/price", r.articleHandler.ChangeArticlePrice) // Slice 2
}
